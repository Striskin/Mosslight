cmake_minimum_required(VERSION 3.20)
project(Mosslight VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(raylib 5.5 QUIET)
if(NOT raylib_FOUND)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(raylib)
endif()
set(MOSS_CORE src/collision.cpp src/combat.cpp src/dialogue.cpp src/enemies.cpp src/game.cpp src/inventory.cpp src/maps.cpp src/player.cpp src/procedural.cpp src/save.cpp)
add_library(moss_core STATIC ${MOSS_CORE})
target_include_directories(moss_core PUBLIC src)
if(MSVC)
    target_compile_options(moss_core PRIVATE /W4)
else()
    target_compile_options(moss_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(Mosslight src/main.cpp src/render.cpp src/ui.cpp src/audio.cpp)
target_link_libraries(Mosslight PRIVATE moss_core raylib)
add_custom_command(TARGET Mosslight POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/data $<TARGET_FILE_DIR:Mosslight>/data)
include(CTest)
if(BUILD_TESTING)
    add_executable(MosslightTests tests/tests.cpp)
    target_link_libraries(MosslightTests PRIVATE moss_core)
    add_test(NAME simulation COMMAND MosslightTests ${CMAKE_SOURCE_DIR}/data ${CMAKE_BINARY_DIR}/test-output)
endif()
